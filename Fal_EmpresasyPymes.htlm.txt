Este fondo sirve para cubrir al trabajador ante cualquier tipo de cese de la relación laboral (despido, renuncia, mutuo acuerdo), brindando previsibilidad al empleador y seguridad al empleado.

        function calcularFAL() {
            let inputEmpleados = document.getElementById('empleados').value;
            let numEmpleados = parseInt(inputEmpleados);

            // Validaciones básicas
            if (isNaN(numEmpleados) || numEmpleados < 1) {
                numEmpleados = 1;
            } else if (numEmpleados > 5000) {
                numEmpleados = 5000;
                document.getElementById('empleados').value = 5000;
            }

            // Variables según categoría
            let categoria = '';
            let sueldoBase = 0;
            let porcentajeAporte = 0;
            let claseBadge = '';

            // Lógica de agrupamiento solicitada
            if (numEmpleados >= 1 && numEmpleados <= 10) {
                categoria = 'MICRO';
                sueldoBase = 1000000; // 1 Millón
                porcentajeAporte = 0.025; // 2.5%
                claseBadge = 'bg-emerald-500';
            } else if (numEmpleados >= 11 && numEmpleados <= 1000) {
                categoria = 'PYME';
                sueldoBase = 1400000; // 1.4 Millones
                porcentajeAporte = 0.025; // 2.5%
                claseBadge = 'bg-blue-500';
            } else if (numEmpleados >= 1001 && numEmpleados <= 5000) {
                categoria = 'GRANDE';
                sueldoBase = 2400000; // 2.4 Millones
                porcentajeAporte = 0.01; // 1%
                claseBadge = 'bg-indigo-600';
            }

            // Actualizar UI del panel de configuración
            const badge = document.getElementById('categoria-badge');
            badge.innerText = categoria;
            badge.className = `px-2 py-1 text-white text-xs font-bold rounded-full ${claseBadge}`;
            
            document.getElementById('sueldo-txt').innerText = formatMoney(sueldoBase);
            document.getElementById('porcentaje-txt').innerText = (porcentajeAporte * 100).toString().replace('.', ',') + '%';
            document.getElementById('res-porcentaje').innerText = (porcentajeAporte * 100).toString().replace('.', ',') + '%';

            // Cálculos matemáticos
            const masaSalarialMensual = numEmpleados * sueldoBase;
            const aporteFALMensual = masaSalarialMensual * porcentajeAporte;
            const capital1Anio = aporteFALMensual * 12;
            const capital3Anios = aporteFALMensual * 36;

            // Actualizar tarjetas de resultados
            document.getElementById('res-masa').innerText = formatMoney(masaSalarialMensual);
            document.getElementById('res-aporte').innerText = formatMoney(aporteFALMensual);
            document.getElementById('res-anio1').innerText = formatMoney(capital1Anio);
            document.getElementById('res-anio3').innerText = formatMoney(capital3Anios);
        }

        // Lógica para cambiar de pestañas
        function switchTab(tabId) {
            // Ocultar todos los contenidos
            document.querySelectorAll('.tab-content').forEach(content => {
                content.classList.remove('active');
            });
            
            // Mostrar el contenido seleccionado
            document.getElementById(tabId).classList.add('active');

            // Resetear estilos de todos los botones
            document.querySelectorAll('.tab-btn').forEach(btn => {
                btn.classList.remove('border-blue-600', 'text-blue-600');
                btn.classList.add('border-transparent', 'text-slate-500');
            });

            // Activar estilo en el botón seleccionado
            const activeBtn = document.querySelector(`.tab-btn[data-target="${tabId}"]`);
            activeBtn.classList.remove('border-transparent', 'text-slate-500');
            activeBtn.classList.add('border-blue-600', 'text-blue-600');
        }

        // Inicializar el cálculo al cargar la página
        window.onload = calcularFAL;
    </script>
